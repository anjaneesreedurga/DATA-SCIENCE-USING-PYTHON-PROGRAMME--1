import numpy as np 
array2d = np.array(  [ [5, 10, 15, 20], 
                       [25, 30, 35, 40], 
                       [45, 50, 55, 60], 
                       [65, 70, 75, 80] ] ) 
print("array2d[1:3, 1:3]:") 
print(array2d[1:3, 1:3]) 
print("array2d[:2, :2]:") 
print(array2d[:2, :2]) 
print("array2d[2:, 2:]:") 
print(array2d[2:, 2:])
